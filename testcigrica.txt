TIME_WINDOWS = {
    "trusova":       (6.0, 10.0),
    "khoreva":       (0.0, 4.0),
    "marianela":     (0.0, 4.0),
    "kapitonova":    (0.0, 4.0),
    "liu":           (0.0, 4.0),
    "valieva":       (0.0, 4.0),
    "kamilavalieva": (0.0, 4.0),

DE_LEVA_FEMALE = {
    "Head":        {"mass": 0.0668, "pos": 0.4841},
    "Trunk":       {"mass": 0.4257, "pos": 0.3782},

if not os.path.exists(INPUT_DIR):
    INPUT_DIR = "skracene_koordinate" if os.path.exists("skracene_koordinate") else "konacne_koordinate"
if not os.path.exists(INPUT_DIR):

# =============================================================================
# 2. BIOMEHANIČKO FILTRIRANJE (BUTTERWORTH ZERO-PHASE)
# =============================================================================

def interp_seg(p_prox, p_dist, ratio):
    return p_prox + ratio * (p_dist - p_prox)

def biomechanical_lowpass(data, cutoff_hz=2.8, fs=30.0, order=4):
    """Zero-phase Butterworth niskopropusni filter za eliminaciju šuma praćenja."""
    nyq = 0.5 * fs
    normal_cutoff = min(cutoff_hz / nyq, 0.99)
    b, a = butter(order, normal_cutoff, btype='low', analog=False)
    # Padlen zaštita za kraće signale
    padlen = min(15, len(data) - 1)
    return filtfilt(b, a, data, padlen=padlen)

def clean_signal(arr, fs=30.0):
    """Kompletno čišćenje signala sa uklanjanjem nan-ova i filtriranjem."""
    arr = np.asarray(arr, dtype=float).copy()
    arr[arr == 0.0] = np.nan
    s = pd.Series(arr)
    filled = s.interpolate(method='linear', limit_direction='both').bfill().ffill().values
    
    # 1. Median filter za impulsni šum (okluzije)
    filled = median_filter(filled, size=3)
    
    # 2. Butterworth filter (eliminiše optički jitter kamere)
    if len(filled) >= 12:
        filled = biomechanical_lowpass(filled, cutoff_hz=3.0, fs=fs)
    return filled

all_files = glob.glob(os.path.join(INPUT_DIR, "*_kinematics.csv"))
if not all_files:
    all_files = glob.glob(os.path.join(INPUT_DIR, "*.csv"))
all_files = [f for f in all_files if "SUMARNA" not in f and "popravljeno" not in f and "fizika_" not in f and "tabela_" not in f]

summary_rows = []
processed_names = set()

print("\n" + "="*145)
print("  FINO PODEŠENA ANALIZA: PRAĆENJE TOPPLE INERCIJE Ix(t), SPIN INERCIJE Iz(t) I STABILNOSTI MASE m(t)")
print("="*145 + "\n")

G_CONST = 9.81  # m/s^2

for file in sorted(all_files):
    df_raw = pd.read_csv(file)
    filename_lower = os.path.basename(file).lower().replace("_", "").replace("-", "")
    athlete_key = next((k for k in ATHLETE_DB if k in filename_lower), None)
    if not athlete_key:
        continue
        
    athlete_data = ATHLETE_DB[athlete_key]
    atype = athlete_data["type"]
    height_m = athlete_data["height"]
    weight_kg = athlete_data["weight"]
    clean_name = athlete_key.upper()

    if clean_name in processed_names:
        continue
    processed_names.add(clean_name)

    if "x_0" not in df_raw.columns and "X_0" not in df_raw.columns:
        continue

    # Kropovanje
    t_start, t_end = TIME_WINDOWS.get(athlete_key, (0.0, 4.0))
    if "timestamp_sec" in df_raw.columns:
        t_arr = df_raw["timestamp_sec"].values
    elif "Time_s" in df_raw.columns:
        t_arr = df_raw["Time_s"].values
    else:
        t_arr = np.arange(len(df_raw)) / 30.0

    mask = (t_arr >= t_start) & (t_arr <= t_end)
    df_crop = df_raw[mask].copy().reset_index(drop=True)
    if len(df_crop) < 15:
        df_crop = df_raw.iloc[:120].copy().reset_index(drop=True)

    n_frames = len(df_crop)
    time_axis = np.linspace(0, 4.0, n_frames)
    dt = 4.0 / max(1, n_frames - 1)
    fs = 1.0 / dt

    # Učitavanje i filtriranje koordinata
    pts_array = np.zeros((n_frames, 33, 3))
    for lm in range(33):
        for ax_idx, ax_name in enumerate(['x', 'y', 'z']):
            col = f"{ax_name}_{lm}" if f"{ax_name}_{lm}" in df_crop.columns else f"{ax_name.upper()}_{lm}"
            if col in df_crop.columns:
                pts_array[:, lm, ax_idx] = clean_signal(df_crop[col].values, fs=fs)

    # Detekcija stajne noge
    var_l = np.median(np.abs(pts_array[:, 31, :2] - np.median(pts_array[:, 31, :2], axis=0)))
    var_r = np.median(np.abs(pts_array[:, 32, :2] - np.median(pts_array[:, 32, :2], axis=0)))
    planted_side = "left" if var_l <= var_r else "right"
    p_ank = 27 if planted_side == "left" else 28
    p_toe = 31 if planted_side == "left" else 32

    # Skaliranje
    mid_shoulder = (pts_array[:, 11, :] + pts_array[:, 12, :]) / 2.0
    mid_hip = (pts_array[:, 23, :] + pts_array[:, 24, :]) / 2.0
    head_vertex = pts_array[:, 0, :] + 0.5 * (pts_array[:, 0, :] - mid_shoulder)
    h_chain = np.linalg.norm(head_vertex - mid_hip, axis=1) + np.linalg.norm(mid_hip - pts_array[:, p_ank, :], axis=1)
    scale = height_m / np.median(h_chain[h_chain > 0.4])
    pts_m = pts_array * scale

    # De Leva segmenti
    mid_sh_m = (pts_m[:, 11, :] + pts_m[:, 12, :]) / 2.0
    mid_hp_m = (pts_m[:, 23, :] + pts_m[:, 24, :]) / 2.0
    head_v_m = pts_m[:, 0, :] + 0.5 * (pts_m[:, 0, :] - mid_sh_m)

    seg_endpoints = {
        "Head":       (mid_sh_m, head_v_m),
        "Trunk":      (mid_sh_m, mid_hp_m),
        "R_UpperArm": (pts_m[:, 12, :], pts_m[:, 14, :]),
        "L_UpperArm": (pts_m[:, 11, :], pts_m[:, 13, :]),
        "R_Forearm":  (pts_m[:, 14, :], pts_m[:, 16, :]),
        "L_Forearm":  (pts_m[:, 13, :], pts_m[:, 15, :]),
        "R_Hand":     (pts_m[:, 16, :], pts_m[:, 16, :] + 0.05 * (pts_m[:, 16, :] - pts_m[:, 14, :])),
        "L_Hand":     (pts_m[:, 15, :], pts_m[:, 15, :] + 0.05 * (pts_m[:, 15, :] - pts_m[:, 13, :])),
        "R_Thigh":    (pts_m[:, 24, :], pts_m[:, 26, :]),
        "L_Thigh":    (pts_m[:, 23, :], pts_m[:, 25, :]),
        "R_Shank":    (pts_m[:, 26, :], pts_m[:, 28, :]),
        "L_Shank":    (pts_m[:, 25, :], pts_m[:, 27, :]),
        "R_Foot":     (pts_m[:, 30, :], pts_m[:, 32, :]),
        "L_Foot":     (pts_m[:, 29, :], pts_m[:, 31, :])
    }

    seg_m = {k: interp_seg(p1, p2, DE_LEVA_FEMALE[k]["pos"]) for k, (p1, p2) in seg_endpoints.items()}

    # Pod
    foot_ground = (pts_m[:, p_ank, :] + pts_m[:, p_toe, :]) / 2.0
    pivot_y_ground = np.max(foot_ground[:, 1])

    # Računanje CoM i Inercija
    com_pos = np.zeros((n_frames, 3))
    Ix_raw = np.zeros(n_frames)
    Iy_raw = np.zeros(n_frames)
    Iz_raw = np.zeros(n_frames)

    for seg_name, s_coords in seg_m.items():
        mass_frac = DE_LEVA_FEMALE[seg_name]["mass"]
        m_seg = mass_frac * weight_kg
        com_pos += mass_frac * s_coords
        
        X_lat = s_coords[:, 0] - mid_hp_m[:, 0]
        Y_ap  = s_coords[:, 2] - mid_hp_m[:, 2]
        Z_vert = np.abs(pivot_y_ground - s_coords[:, 1])

        Ix_raw += m_seg * (Y_ap**2 + Z_vert**2)
        Iy_raw += m_seg * (X_lat**2 + Z_vert**2)
        Iz_raw += m_seg * (X_lat**2 + Y_ap**2)

    # Filtriranje tenzora inercije
    win_sg = max(7, min(17, n_frames - 1 | 1))
    Ix = savgol_filter(Ix_raw, window_length=win_sg, polyorder=2)
    Iy = savgol_filter(Iy_raw, window_length=win_sg, polyorder=2)
    Iz = savgol_filter(Iz_raw, window_length=win_sg, polyorder=2)

    # -------------------------------------------------------------------------
    # 3. KOREKTAN I STABILIZOVAN PRORAČUN DINAMIČKE MASE m(t)
    # -------------------------------------------------------------------------
    # Visina centra mase iznad poda
    com_z = np.abs(pivot_y_ground - com_pos[:, 1])
    
    # Biomehaničko filtriranje CoM trajektorije na 2.2 Hz (fizički pokret tela)
    com_z_filtered = biomechanical_lowpass(com_z, cutoff_hz=2.2, fs=fs)
    
    # Prvi i drugi izvod (ubrzanje)
    vel_z = np.gradient(com_z_filtered, dt)
    acc_z = np.gradient(vel_z, dt)
    
    # Još jedan prolaz niskopropusnog filtera na ubrzanju da se eliminiše preostali šum diferenciranja
    acc_z_clean = biomechanical_lowpass(acc_z, cutoff_hz=2.0, fs=fs)
    
    # Realno biomehaničko fizičko ograničenje ubrzanja u rotaciji (max ±0.20 g)
    acc_z_clean = np.clip(acc_z_clean, -0.20 * G_CONST, 0.20 * G_CONST)

    # Dinamička prividna masa: m(t) = m0 * (1 + a_z / g)
    m_dynamic = weight_kg * (1.0 + acc_z_clean / G_CONST)

    m_min = np.min(m_dynamic)
    m_max = np.max(m_dynamic)
    m_mean = np.mean(m_dynamic)
    delta_pct = ((m_max - m_min) / (2.0 * weight_kg)) * 100.0

    summary_rows.append({
        "Sportista": clean_name,
        "Tip": atype,
        "Stvarna masa m0 [kg]": weight_kg,
        "Ix sr [kg*m2]": round(np.mean(Ix), 2),
        "m(t) min [kg]": round(m_min, 2),
        "m(t) max [kg]": round(m_max, 2),
        "m(t) sr [kg]": round(m_mean, 2),
        "Varijacija mase Delta [%]": f"±{delta_pct:.1f}%",
        "Zaključak modela": "Masa stabilna (Ix ≈ const)"
    })

    # =========================================================================
    # 4. GENERISANJE GRAFIKA
    # =========================================================================
    fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(15.5, 5.8), facecolor='#0b0f19')
    for ax in (ax1, ax2):
        ax.set_facecolor('#0b0f19')
        ax.grid(True, linestyle='--', alpha=0.35, color='#1e293b')
        ax.tick_params(colors='#94a3b8', labelsize=9.5)

    ax1.plot(time_axis, m_dynamic, color='#38bdf8', linewidth=2.4, label='m(t) dinamička prividna masa')
    ax1.axhline(weight_kg, color='#f43f5e', linestyle='--', linewidth=2.0, label=f'Stvarna masa m0 = {weight_kg:.1f} kg')
    ax1.fill_between(time_axis, m_min, m_max, color='#38bdf8', alpha=0.12, label=f'Opseg: [{m_min:.1f} - {m_max:.1f}] kg')
    ax1.set_xlabel("Vreme [s]", fontsize=10.5, fontweight='bold', color='#94a3b8')
    ax1.set_ylabel("Masa [kg]", fontsize=10.5, fontweight='bold', color='#94a3b8')
    ax1.set_ylim(weight_kg * 0.7, weight_kg * 1.3)
    ax1.set_title(f"ZAVISNOST DINAMIČKE MASE OD VREMENA m(t)\nStabilizovan Butterworth filter (Realna varijacija)", fontsize=11.5, fontweight='bold', color='#38bdf8', pad=12)
    ax1.legend(loc='upper right', facecolor='#111827', edgecolor='#374151', labelcolor='white', fontsize=8.5)

    ax2.plot(time_axis, Ix, color='#fb7185', linewidth=2.2, label='Ix(t) topple oko poda (sagitalno)')
    ax2.plot(time_axis, Iy, color='#facc15', linewidth=2.0, linestyle='--', label='Iy(t) topple oko poda (frontalno)')
    ax2.plot(time_axis, Iz, color='#00f5d4', linewidth=2.2, label='Iz(t) spin oko kičme')
    ax2.set_xlabel("Vreme [s]", fontsize=10.5, fontweight='bold', color='#94a3b8')
    ax2.set_ylabel("Moment inercije [kg·m²]", fontsize=10.5, fontweight='bold', color='#94a3b8')
    ax2.set_title(f"MOMENTI INERCIJE Ix, Iy (TOPPLE) I Iz (SPIN)", fontsize=11.5, fontweight='bold', color='#fb7185', pad=12)
    ax2.legend(loc='upper right', facecolor='#111827', edgecolor='#374151', labelcolor='white', fontsize=8.5)

    plt.suptitle(f"DINAMIKA I VERIFIKACIJA MASE: {clean_name} ({atype})", fontsize=13.5, fontweight='bold', color='#ffffff', y=0.98)
    plt.tight_layout()
    plt.savefig(os.path.join(OUT_DIR, f"zavisnost_mase_i_inercije_{clean_name.lower()}.png"), dpi=300, facecolor='#0b0f19')
    plt.close()

    print(f"[OBRAĐENO] {clean_name:<14} | Ix sr: {np.mean(Ix):5.2f} kg*m² | m_real: {weight_kg:4.1f} kg | m(t) opseg: [{m_min:4.1f} - {m_max:4.1f}] kg | Varijacija: ±{delta_pct:4.1f}%")

# =============================================================================
# 5. TABELA I IZVOZ
# =============================================================================
df_res = pd.DataFrame(summary_rows)
df_res.to_csv(os.path.join(OUT_DIR, "tabela_stabilnosti_mase_i_inercije.csv"), index=False)

print("\n" + "="*145)
print("  FINALNA TABELA: PROVERA ZAVISNOSTI MASE OD VREMENA IZ REALNOG TOPPLE MODELA")
print("="*145)
print(df_res.to_string(index=False))
print("="*145 + "\n")
print(f"✓ Rezultati i grafici su sačuvani u direktorijumu: '{OUT_DIR}/'")
