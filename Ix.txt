    "kapitonova":   {"height": 1.68, "weight": 48.0, "type": "Balet"},

    "scerebakova":  {"height": 1.61, "weight": 42.0, "type": "Umetničko klizanje"},
    "liu":          {"height": 1.58, "weight": 45.0, "type": "Umetničko klizanje"}
}

TIME_WINDOWS = {
    "trusova":       (6.0, 10.0),
    "khoreva":       (0.0, 4.0),
    "marianela":     (0.0, 4.0),
    "kapitonova":    (0.0, 4.0),
    "liu":           (0.0, 4.0),
    "valieva":       (0.0, 4.0),
    "kamilavalieva": (0.0, 4.0),
    "shcherbakova":  (0.0, 4.0),
    "scerebakova":   (0.0, 4.0)
}

DE_LEVA_FEMALE = {
    "Head":        {"mass": 0.0668, "pos": 0.4841},
    "Trunk":       {"mass": 0.4257, "pos": 0.3782},
    "R_UpperArm":  {"mass": 0.0255, "pos": 0.5754},
    "L_UpperArm":  {"mass": 0.0255, "pos": 0.5754},
    "R_Forearm":   {"mass": 0.0138, "pos": 0.4559},

    "L_Shank":     {"mass": 0.0481, "pos": 0.4416},
    "R_Foot":      {"mass": 0.0129, "pos": 0.4014},
    "L_Foot":      {"mass": 0.0129, "pos": 0.4014}
}

INPUT_DIR = "kinematika_rezultati"
if not os.path.exists(INPUT_DIR):
    INPUT_DIR = "skracene_koordinate" if os.path.exists("skracene_koordinate") else "konacne_koordinate"
if not os.path.exists(INPUT_DIR):
    INPUT_DIR = "."

OUT_DIR = "konacnirezultati/verifikacija_topple_inercije"
os.makedirs(OUT_DIR, exist_ok=True)

# =============================================================================
# 2. NUMERIČKE I POMOĆNE FUNKCIJE
# =============================================================================

def interp_seg(p_prox, p_dist, ratio):
    return p_prox + ratio * (p_dist - p_prox)

def clean_signal(arr):
    arr = np.asarray(arr, dtype=float).copy()
    arr[arr == 0.0] = np.nan
    s = pd.Series(arr)
    filled = s.interpolate(method='linear', limit_direction='both').bfill().ffill().values
    filled = median_filter(filled, size=3)
    win = 9 if len(filled) >= 9 else (len(filled) if len(filled) % 2 != 0 else len(filled) - 1)
    if win >= 5:
        filled = savgol_filter(filled, window_length=win, polyorder=2)
    return filled

all_files = glob.glob(os.path.join(INPUT_DIR, "*_kinematics.csv"))
if not all_files:
    all_files = glob.glob(os.path.join(INPUT_DIR, "*.csv"))
all_files = [f for f in all_files if "SUMARNA" not in f and "popravljeno" not in f and "fizika_" not in f]

summary_rows = []
processed_names = set()

print("\n" + "="*135)
print("  PRORAČUN 3D TENZORA INERCIJE OKO PODA: Ix (TOPPLE), Iy (TOPPLE) I Iz (SPIN)")
print("="*135 + "\n")

for file in sorted(all_files):
    df_raw = pd.read_csv(file)
    filename_lower = os.path.basename(file).lower().replace("_", "").replace("-", "")
    athlete_key = next((k for k in ATHLETE_DB if k in filename_lower), None)
    if not athlete_key:
        continue
        
    athlete_data = ATHLETE_DB[athlete_key]
    atype = athlete_data["type"]
    height_m = athlete_data["height"]
    weight_kg = athlete_data["weight"]
    clean_name = athlete_key.upper()

    if clean_name in processed_names:
        continue
    processed_names.add(clean_name)

    if "x_0" not in df_raw.columns and "X_0" not in df_raw.columns:
        continue

    # Kropovanje vremenskog intervala
    t_start, t_end = TIME_WINDOWS.get(athlete_key, (0.0, 4.0))
    if "timestamp_sec" in df_raw.columns:
        t_arr = df_raw["timestamp_sec"].values
    elif "Time_s" in df_raw.columns:
        t_arr = df_raw["Time_s"].values
    else:
        t_arr = np.arange(len(df_raw)) / 30.0

    mask = (t_arr >= t_start) & (t_arr <= t_end)
    df_crop = df_raw[mask].copy().reset_index(drop=True)
    if len(df_crop) < 15:
        df_crop = df_raw.iloc[:120].copy().reset_index(drop=True)

    n_frames = len(df_crop)
    time_axis = np.linspace(0, 4.0, n_frames)

    # Učitavanje 3D MediaPipe koordinata
    pts_array = np.zeros((n_frames, 33, 3))
    for lm in range(33):
        for ax_idx, ax_name in enumerate(['x', 'y', 'z']):
            col = f"{ax_name}_{lm}" if f"{ax_name}_{lm}" in df_crop.columns else f"{ax_name.upper()}_{lm}"
            if col in df_crop.columns:
                pts_array[:, lm, ax_idx] = clean_signal(df_crop[col].values)

    # Detekcija stajne noge (pivot)
    var_l = np.median(np.abs(pts_array[:, 31, :2] - np.median(pts_array[:, 31, :2], axis=0)))
    var_r = np.median(np.abs(pts_array[:, 32, :2] - np.median(pts_array[:, 32, :2], axis=0)))
    planted_side = "left" if var_l <= var_r else "right"
    p_ank = 27 if planted_side == "left" else 28
    p_toe = 31 if planted_side == "left" else 32

    # Metričko skaliranje
    mid_shoulder = (pts_array[:, 11, :] + pts_array[:, 12, :]) / 2.0
    mid_hip = (pts_array[:, 23, :] + pts_array[:, 24, :]) / 2.0
    head_vertex = pts_array[:, 0, :] + 0.5 * (pts_array[:, 0, :] - mid_shoulder)
    h_chain = np.linalg.norm(head_vertex - mid_hip, axis=1) + np.linalg.norm(mid_hip - pts_array[:, p_ank, :], axis=1)
    scale = height_m / np.median(h_chain[h_chain > 0.4])
    pts_m = pts_array * scale

    # Segmenti De Leva
    mid_sh_m = (pts_m[:, 11, :] + pts_m[:, 12, :]) / 2.0
    mid_hp_m = (pts_m[:, 23, :] + pts_m[:, 24, :]) / 2.0
    head_v_m = pts_m[:, 0, :] + 0.5 * (pts_m[:, 0, :] - mid_sh_m)

    seg_endpoints = {
        "Head":       (mid_sh_m, head_v_m),
        "Trunk":      (mid_sh_m, mid_hp_m),
        "R_UpperArm": (pts_m[:, 12, :], pts_m[:, 14, :]),
        "L_UpperArm": (pts_m[:, 11, :], pts_m[:, 13, :]),
        "R_Forearm":  (pts_m[:, 14, :], pts_m[:, 16, :]),
        "L_Forearm":  (pts_m[:, 13, :], pts_m[:, 15, :]),
        "R_Hand":     (pts_m[:, 16, :], pts_m[:, 16, :] + 0.05 * (pts_m[:, 16, :] - pts_m[:, 14, :])),
        "L_Hand":     (pts_m[:, 15, :], pts_m[:, 15, :] + 0.05 * (pts_m[:, 15, :] - pts_m[:, 13, :])),
        "R_Thigh":    (pts_m[:, 24, :], pts_m[:, 26, :]),
        "L_Thigh":    (pts_m[:, 23, :], pts_m[:, 25, :]),
        "R_Shank":    (pts_m[:, 26, :], pts_m[:, 28, :]),
        "L_Shank":    (pts_m[:, 25, :], pts_m[:, 27, :]),
        "R_Foot":     (pts_m[:, 30, :], pts_m[:, 32, :]),
        "L_Foot":     (pts_m[:, 29, :], pts_m[:, 31, :])
    }

    seg_m = {k: interp_seg(p1, p2, DE_LEVA_FEMALE[k]["pos"]) for k, (p1, p2) in seg_endpoints.items()}

    # -------------------------------------------------------------------------
    # TAČKA OSLONCA NA PODU (PIVOT = ISHODIŠTE KOORDINATA)
    # -------------------------------------------------------------------------
    foot_ground = (pts_m[:, p_ank, :] + pts_m[:, p_toe, :]) / 2.0
    pivot_x = np.median(foot_ground[:, 0])
    pivot_y_ground = np.max(foot_ground[:, 1])  # Nivo poda (najniža Y tačka MediaPipe-a)
    pivot_z = np.median(foot_ground[:, 2])

    # -------------------------------------------------------------------------
    # 3D PRORAČUN: Ix (TOPPLE), Iy (TOPPLE), Iz (SPIN)
    # -------------------------------------------------------------------------
    Ix_raw = np.zeros(n_frames)  # Topple oko horizontalne ose X
    Iy_raw = np.zeros(n_frames)  # Topple oko horizontalne ose Y (anteroposteriorno)
    Iz_raw = np.zeros(n_frames)  # Spin oko uspravne ose Z (tvoj I_B)

    for seg_name, s_coords in seg_m.items():
        m_seg = DE_LEVA_FEMALE[seg_name]["mass"] * weight_kg
        
        # 1. Horizontalna rastojanja od uspravne kičme:
        X_lat = s_coords[:, 0] - mid_hp_m[:, 0]  # Lateralno odstupanje [m]
        Y_ap  = s_coords[:, 2] - mid_hp_m[:, 2]  # Anteroposteriorno odstupanje [m]
        
        # 2. Vertikalna visina iznad poda:
        Z_vert = np.abs(pivot_y_ground - s_coords[:, 1])  # Visina iznad poda [m]

        # A) TOPPLE Ix: Rotacija u ravni Y-Z (oko horizontalne X ose) -> ulazi visina Z_vert^2
        Ix_raw += m_seg * (Y_ap**2 + Z_vert**2)

        # B) TOPPLE Iy: Rotacija u ravni X-Z (oko horizontalne Y ose) -> ulazi visina Z_vert^2
        Iy_raw += m_seg * (X_lat**2 + Z_vert**2)

        # C) SPIN Iz: Rotacija u ravni X-Y (oko uspravne kičme) -> samo horizontalni krakovi
        Iz_raw += m_seg * (X_lat**2 + Y_ap**2)

    win = max(7, min(15, n_frames - 1 | 1))
    Ix = savgol_filter(Ix_raw, window_length=win, polyorder=2)
    Iy = savgol_filter(Iy_raw, window_length=win, polyorder=2)
    Iz = savgol_filter(Iz_raw, window_length=win, polyorder=2)

    # Skalirana Lott & Laws referenca za ovu visinu i masu
    ref_scale = (weight_kg / 45.0) * ((height_m / 1.50) ** 2)
    lott_ref_topple = 55.3 * ref_scale
    lott_ref_spin = 0.534 * ref_scale

    summary_rows.append({
        "Sportista": clean_name,
        "Tip": atype,
        "Visina [m]": height_m,
        "Masa [kg]": weight_kg,
        "Ix (Topple oko poda) [kg*m2]": round(np.mean(Ix), 2),
        "Iy (Topple oko poda) [kg*m2]": round(np.mean(Iy), 2),
        "Iz (Spin oko kičme) [kg*m2]": round(np.mean(Iz), 2),
        "Lott & Laws Ref Ix/Iy": round(lott_ref_topple, 2),
        "Lott & Laws Ref Iz": round(lott_ref_spin, 3)
    })

    # Pojedinačni grafik poređenja Ix, Iy i Iz
    fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(15.5, 6.2), facecolor='#0b0f19')
    for ax in (ax1, ax2):
        ax.set_facecolor('#0b0f19')
        ax.grid(True, linestyle='--', alpha=0.35, color='#1e293b')
        ax.tick_params(colors='#94a3b8', labelsize=9.5)

    # Leva strana: TOPPLE momenti inercije oko poda (Ix i Iy)
    ax1.plot(time_axis, Ix, color='#fb7185', linewidth=2.4, label='Ix(t) topple oko poda (sagitalno)')
    ax1.plot(time_axis, Iy, color='#facc15', linewidth=2.2, linestyle='--', label='Iy(t) topple oko poda (frontalno)')
    ax1.axhline(lott_ref_topple, color='#38bdf8', linestyle=':', linewidth=1.8, label=f'Lott & Laws skalirana ref: {lott_ref_topple:.1f} kg·m²')
    ax1.set_xlabel("Vreme [s]", fontsize=10.5, fontweight='bold', color='#94a3b8')
    ax1.set_ylabel("Topple moment inercije [kg·m²]", fontsize=10.5, fontweight='bold', color='#94a3b8')
    ax1.set_title(f"TOPPLE INERCIJA OKO PODA (Ix i Iy)\nVisina Z_vert ulazi u formulu (Ogroman otpor padu)", fontsize=11.5, fontweight='bold', color='#fb7185', pad=12)
    ax1.legend(loc='upper right', facecolor='#111827', edgecolor='#374151', labelcolor='white', fontsize=8.5)

    # Desna strana: SPIN moment inercije oko uspravne ose (Iz)
    ax2.plot(time_axis, Iz, color='#00f5d4', linewidth=2.4, label='Iz(t) spin oko uspravne kičme (tvoj I_B)')
    ax2.axhline(np.mean(Iz), color='#a78bfa', linestyle='--', linewidth=1.6, label=f'Srednji Iz: {np.mean(Iz):.2f} kg·m²')
    ax2.axhline(lott_ref_spin, color='#38bdf8', linestyle=':', linewidth=1.6, label=f'Lott & Laws ref Iz: {lott_ref_spin:.3f} kg·m²')
    ax2.set_xlabel("Vreme [s]", fontsize=10.5, fontweight='bold', color='#94a3b8')
    ax2.set_ylabel("Spin moment inercije [kg·m²]", fontsize=10.5, fontweight='bold', color='#94a3b8')
    ax2.set_title(f"SPIN INERCIJA OKO KIČME (Iz = tvoj I_B)\nSamo horizontalni krakovi ulaze u formulu (Mali otpor)", fontsize=11.5, fontweight='bold', color='#00f5d4', pad=12)
    ax2.legend(loc='upper right', facecolor='#111827', edgecolor='#374151', labelcolor='white', fontsize=8.5)

    plt.suptitle(f"POREĐENJE 3D MOMENATA INERCIJE: {clean_name} ({atype})", fontsize=13.5, fontweight='bold', color='#ffffff', y=0.98)
    plt.tight_layout()
    plt.savefig(os.path.join(OUT_DIR, f"inercija_3d_komparacija_{clean_name.lower()}.png"), dpi=300, facecolor='#0b0f19')
    plt.close()

    print(f"[OBRAĐENO] {clean_name:<14} | Ix (topple): {np.mean(Ix):5.2f} kg*m² | Iy (topple): {np.mean(Iy):5.2f} kg*m² | Iz (spin): {np.mean(Iz):4.2f} kg*m²")

# =============================================================================
# 3. TABELA I IZVOZ REZULTATA
# =============================================================================
df_res = pd.DataFrame(summary_rows)
df_res.to_csv(os.path.join(OUT_DIR, "tabela_3d_momenti_inercije_ix_iy_iz.csv"), index=False)

print("\n" + "="*150)
print("  FINALNA TABELA: POREĐENJE TOPPLE MOMENATA (Ix, Iy) I SPIN MOMENTA (Iz) SA LOTT & LAWS REFERENCOM")
print("="*150)
print(df_res.to_string(index=False))
print("="*150 + "\n")
print(f"✓ Grafici i tabela su sačuvani u: '{OUT_DIR}/'")
